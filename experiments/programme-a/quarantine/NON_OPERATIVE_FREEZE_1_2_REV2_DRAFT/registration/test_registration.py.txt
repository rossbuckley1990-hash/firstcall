class NetworkBan(unittest.TestCase):
    def setUp(self):
        self._orig = (socket.socket.connect, socket.create_connection, socket.getaddrinfo)

        def banned(*a, **k):
            raise AssertionError("network access attempted during offline test")
        socket.socket.connect = banned
        socket.create_connection = banned
        socket.getaddrinfo = banned

    def tearDown(self):
        socket.socket.connect, socket.create_connection, socket.getaddrinfo = self._orig


def gh(n, total=None, incomplete=False, owner_type="Organization"):
    total = n if total is None else total
    items = [{"id": 1000 + i, "full_name": f"org{i}/catalog{i}", "html_url": f"https://example.invalid/org{i}/catalog{i}",
              "owner": {"login": f"person{i}", "type": owner_type, "avatar_url": "https://example.invalid/a.png"},
              "fork": False, "archived": False, "created_at": "2020-01-01T00:00:00Z",
              "pushed_at": "2026-01-01T00:00:00Z", "default_branch": "main",
              "license": {"spdx_id": "CC0-1.0"}, "stargazers_count": 5, "description": "synthetic"}
             for i in range(1, n + 1)]
    return json.dumps({"total_count": total, "incomplete_results": incomplete, "items": items}).encode()


AUP = ("<html><body><p>You may use information from our Service for the following reasons, "
       "<em>regardless of whether the information was scraped, collected through our API, or obtained otherwise</em>:</p>"
       "<ul><li>Researchers may use public, non-personal information from the Service for research purposes, "
       "only if any publications resulting from that research are open access <span>.</span></li></ul>"
       "<p>Scraping does not refer to the collection of information through our API.</p></body></html>").encode()


class FakeEnv:
    def __init__(self, t, responses=None, sntp=(0.0, 0.0), egress=("192.0.2.1", "192.0.2.1"),
                 reg_ok=True, inject_late=None, terms=AUP, robots=(404, b"{}")):
        self.t = t
        self.responses = list(responses if responses is not None else [cr.Response(200, [], gh(20, 500))] * 4)
        self.sntp = list(sntp)
        self.egress = list(egress)
        self.reg_ok = reg_ok
        self.inject_late = inject_late
        self.terms = terms
        self.robots = robots
        self.armed = None
        self.source_calls = []
        self.aux_calls = []

    def now(self):
        return self.t

    def sleep(self, s):
        self.t += max(0.0, s)
        if self.inject_late and self.t >= self.inject_late[0]:
            self.t += self.inject_late[1]
            self.inject_late = None

    def arm_source(self, until):
        self.armed = until

    def disarm_source(self):
        self.armed = None

    def http_get(self, url, headers, timeout):
        if cr.is_source_url(url):
            if self.armed is None or self.t > self.armed:
                raise cr.SourceGuardError("unarmed")
            self.source_calls.append((self.t, url, headers))
            r = self.responses.pop(0)
            if isinstance(r, Exception):
                raise r
            return r
        raise AssertionError("unexpected http " + url)

    def aux_get(self, url):
        self.aux_calls.append((self.t, url))
        if cr.is_source_url(url):
            raise AssertionError("aux path reached discovery endpoint")
        if url == cr.ROBOTS_URL:
            if isinstance(self.robots, Exception):
                raise self.robots
            return cr.Response(self.robots[0], [], self.robots[1])
        if isinstance(self.terms, Exception):
            raise self.terms
        return cr.Response(200, [], self.terms)

    def sntp_offset(self):
        v = self.sntp.pop(0) if self.sntp else 0.0
        if isinstance(v, Exception):
            raise v
        return {"offset_seconds": v, "server": "fake"}

    def egress_ip(self):
        return self.egress.pop(0)

    def verify_registration(self):
        return {"ok": self.reg_ok, "checks": {"fake": self.reg_ok}}

    def keep_awake(self):
        return {"fake": True}

    def runtime_facts(self):
        return {"fake": True}

    def test_happy_path_exact_times_custody_before_projection(self):
        env = FakeEnv(START - 3600)
        m, tmp = self.run_env(env)
        self.assertEqual(m["status"], "CAPTURED_COMPLETE")
        self.assertEqual([round(c[0] - START, 6) for c in env.source_calls], [0, 60, 120, 180])
        self.assertTrue(all(t < START for t, _ in env.aux_calls))
        for i in range(1, 5):
            self.assertEqual(cr.sha256((tmp / f"custody/raw/q{i}.json").read_bytes()), m["ordered_raw_sha256"][i - 1])
            self.assertFalse((tmp / f"cap/raw/q{i}.json").exists())
        events = [(e["event"], e.get("index")) for e in map(json.loads, (tmp / "cap/events.jsonl").read_text().splitlines())]
        for i in range(1, 5):
            self.assertLess(events.index(("response_archived", i)), events.index(("response_projected", i)))
        self.assertLess(events.index(("policies_verified", None)), events.index(("request_due", 1)))
        proj = json.loads((tmp / "cap/projection/q1.json").read_bytes())
        self.assertEqual(len(proj["items"]), 20)
        published = b"".join(p.read_bytes() for p in (tmp / "cap").rglob("*") if p.is_file())
        for personal in (b"person1", b"avatar_url", b"stargazers"):
            self.assertNotIn(personal, published)
        data = (tmp / "cap/session-manifest.json").read_bytes()
        self.assertEqual(data, cr.canonical(json.loads(data)))

def make_repo(files):
    d = Path(tempfile.mkdtemp())
    subprocess.run(["git", "init", "-q", str(d)], check=True)
    for path, text in files.items():
        p = d / path
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_text(text)
    subprocess.run(["git", "-C", str(d), "add", "-A"], check=True)
    subprocess.run(["git", "-C", str(d), "-c", "user.name=t", "-c", "user.email=t@example.invalid",
                    "commit", "-q", "-m", "fixture"], check=True)
    return d, subprocess.run(["git", "-C", str(d), "rev-parse", "HEAD"], check=True,
                             capture_output=True).stdout.decode().strip()


class ProjectorTests(unittest.TestCase):
    def test_grammar_and_exclusions(self):
        repo, commit = make_repo({
            "experiments/x/config.json": json.dumps({"experiment": "EXP-9", "vendor": "Alpha Mail",
                                                     "firstcall_operated": True,
                                                     "runner": "firstcall/run_alpha.py"}),
            "experiments/x/receipt.json": json.dumps({"experiment": "EXP-9", "vendor": "Leaky", "grade": "FAIL"}),
            "experiments/y/plain.json": json.dumps({"vendor_id": "Beta"}),
            "docs/design.md": "Experiment: EXP-10\n\n## 2. Targets\n\n| Vendor | Result |\n|---|---|\n| Gamma | SECRET |\n\n## Results\nVendor: NotThis\n",
            "docs/other.md": "# Target selection\ntext\n",
            "firstcall/run_alpha.py": "print('never executed')\n",
            "artifacts/z.json": json.dumps({"experiment": "E", "vendor": "Out"}),
        })
        r = pj.project(repo, commit)
        self.assertEqual([e["path"] for e in r["ledger"]], sorted((e["path"] for e in r["ledger"]), key=str.encode))
        pos = {(d["path"], d["identity"]) for d in r["positive_json_designations"]}
        self.assertEqual(pos, {("experiments/x/config.json", "Alpha Mail")})
        self.assertTrue(r["positive_json_designations"][0]["firstcall_operated"])
        md = r["positive_markdown_designations"]
        self.assertEqual([(d["path"], d["vendor_column_values"]) for d in md], [("docs/design.md", ["Gamma"])])
        dumped = json.dumps(r)
        for secret in ("SECRET", "Leaky", "NotThis", "FAIL", "Out"):
            self.assertNotIn(secret, dumped)
        reasons = {(u["path"], u["reason"]) for u in r["unrecognized_designations"]}
        self.assertIn(("experiments/y/plain.json", "IDENTITY_KEY_WITHOUT_SAME_LEVEL_EXPERIMENT"), reasons)
        self.assertIn(("docs/other.md", "NON_GRAMMAR_TARGET_HEADING"), reasons)
        self.assertEqual(r["status"], "BLOCKED_UNRECOGNIZED_DESIGNATION")
        cls = {e["path"]: e["class"] for e in r["ledger"]}
        self.assertEqual(cls["experiments/x/receipt.json"], "EXCLUDED_RESULT_OR_EVIDENCE_CONTEXT")
        self.assertEqual(cls["artifacts/z.json"], "OUT_OF_PROJECTION_SCOPE_S8_4")
        self.assertEqual(cls["firstcall/run_alpha.py"], "CODE_PATH_ONLY")
        cfg = next(e for e in r["ledger"] if e["path"] == "experiments/x/config.json")
        self.assertEqual(cfg["literal_code_path_references"], ["firstcall/run_alpha.py"])
        self.assertEqual(r["adjudicator_approvals"], {"ADJ-1": None, "ADJ-2": None})

    def test_hostile_designation_cases(self):
        repo, commit = make_repo({
            # incidental vendor mention in a config (not a target key)
            "experiments/a/config.json": json.dumps({"experiment": "A-1", "notes": "compared against VendorNotes"}),
            # policy field: boolean/null vendor-named keys cannot carry an identity
            "experiments/b/policy.json": json.dumps({"experiment": "B-1", "independent_vendor_verification_required": True,
                                                     "target_timeout_override": None}),
            # actual experiment target
            "experiments/c/manifest.json": json.dumps({"experiment_id": "C-1", "target_vendor": "RealTarget"}),
            # renamed experiment (id string records history) is still positive
            "experiments/d/manifest.json": json.dumps({"experiment": "D-2 (formerly D-1)", "vendor_id": "RenamedTarget"}),
            # nested experiment: identity below the experiment object is not grammar -> blocks
            "experiments/e/manifest.json": json.dumps({"experiment": "E-1", "runs": [{"vendor": "NestedTarget"}]}),
            # policy-like key holding an identity-capable value -> blocks
            "experiments/f/policy.json": json.dumps({"experiment": "F-1", "provider_under_test": "HiddenTarget"}),
            # documentation example in a fenced block and in an example path
            "docs/guide.md": "Intro.\n\n```\nExperiment: X-1\n## Targets\nVendor: FencedExample\n```\n",
            "experiments/examples/sample.json": json.dumps({"experiment": "X-2", "vendor": "ExampleVendor"}),
            # vendor appearing in prose only
            "docs/prose.md": "# Notes\nWe read about Vendor: ProseOnly in passing and ProseTwo elsewhere.\n",
            # historical artifact and receipt contexts
            "artifacts/old/manifest.json": json.dumps({"experiment": "H-1", "vendor": "ArtifactVendor"}),
            "experiments/h/receipt.json": json.dumps({"experiment": "H-2", "vendor": "ReceiptVendor"}),
        })
        r = pj.project(repo, commit)
        pos = {(d["path"], d["identity"]) for d in r["positive_json_designations"]}
        self.assertEqual(pos, {("experiments/c/manifest.json", "RealTarget"),
                               ("experiments/d/manifest.json", "RenamedTarget")})
        self.assertEqual(r["positive_markdown_designations"], [])
        reasons = {(u["path"], u["reason"]) for u in r["unrecognized_designations"]}
        self.assertEqual(reasons, {
            ("experiments/e/manifest.json", "IDENTITY_KEY_WITHOUT_SAME_LEVEL_EXPERIMENT"),
            ("experiments/f/policy.json", "EXPERIMENT_OBJECT_WITH_NON_GRAMMAR_TARGET_KEY"),
            ("experiments/examples/sample.json", "DESIGNATION_SHAPE_IN_EXAMPLE_PATH"),
        })
        b = next(e for e in r["ledger"] if e["path"] == "experiments/b/policy.json")
        self.assertEqual([f["key"] for f in b["projection"]["incidental_non_identity_fields"]],
                         ["independent_vendor_verification_required", "target_timeout_override"])
        dumped = json.dumps(r)
        for leaked in ("VendorNotes", "FencedExample", "ExampleVendor", "ProseOnly", "ProseTwo",
                       "ArtifactVendor", "ReceiptVendor", "NestedTarget", "HiddenTarget"):
            self.assertNotIn(leaked, dumped)

    def test_clean_repo_is_draft_not_approved(self):
        repo, commit = make_repo({"experiments/x/config.json": json.dumps({"experiment": "E", "vendor": "A"})})
        r = pj.project(repo, commit)
        self.assertEqual(r["status"], "DRAFT_REQUIRES_TWO_ADJUDICATOR_APPROVAL")

    def test_deterministic(self):
        repo, commit = make_repo({"docs/a.md": "x", "experiments/b.json": "{}"})
        self.assertEqual(pj.canonical(pj.project(repo, commit)), pj.canonical(pj.project(repo, commit)))

            "functions": sorted(vr.FUNCTIONS), "attestations": {k: True for k in vr.ATTESTATION_KEYS},
            "disclosures": {k: False for k in vr.DISCLOSURE_KEYS}, "conflicts": []}


class ValidatorTests(unittest.TestCase):
    def setUp(self):
        self.roles = json.loads((HERE / "roles.json").read_text())

    def fill_all(self, roles):
        for r in roles["roles"]:
            if r["status"] == "UNFILLED" and r["id"] not in ("R06", "R07", "R14"):
                r["status"] = "FILLED"
                r["registrant"] = {"name": f"Person {r['id']}"}
        return roles

    def test_current_package_is_incomplete_not_ready(self):
        errors, st = vr.evaluate_roles(self.roles, {})
        self.assertEqual(errors, [])
        self.assertEqual(st["decision"], "SEALABLE_BUT_REGISTRATIONS_INCOMPLETE")
        self.assertTrue(st["capture_roles_filled"])
        for rid in ("R06", "R07", "R08", "R09", "R10", "R15", "R16"):
            self.assertIn(rid, st["unfilled"])

    def test_ready_only_when_everything_registrable_is_filled(self):
        roles = self.fill_all(copy.deepcopy(self.roles))
        adj = {"ADJ-1": adjudicator("ADJ-1", "A Person", "a@example.invalid"),
               "ADJ-2": adjudicator("ADJ-2", "B Person", "b@example.invalid")}
        errors, st = vr.evaluate_roles(roles, adj)
        self.assertEqual(errors, [])
        self.assertEqual(st["decision"], "SEALABLE_AND_CAPTURE_READY")
        errors, st = vr.evaluate_roles(roles, {"ADJ-1": adj["ADJ-1"]})
        self.assertEqual(st["decision"], "SEALABLE_BUT_REGISTRATIONS_INCOMPLETE")

    def test_same_person_twice_is_not_independent(self):
        adj = {"ADJ-1": adjudicator("ADJ-1", "Same  Person", "x@example.invalid"),
               "ADJ-2": adjudicator("ADJ-2", "same person", "y@example.invalid")}
        errors, st = vr.evaluate_roles(self.roles, adj)
        self.assertEqual(st["decision"], "NOT_SEALABLE")

    def test_model_adjudicator_rejected(self):
        rec = adjudicator("ADJ-1", "Claude instance A", "n/a")
        rec["kind"] = "model"
        errors, st = vr.evaluate_roles(self.roles, {"ADJ-1": rec})
        self.assertTrue(any("natural_person" in e for e in errors))
        self.assertEqual(st["decision"], "NOT_SEALABLE")

    def test_missing_attestation_rejected(self):
        rec = adjudicator("ADJ-1", "A", "a@example.invalid")
        rec["attestations"].pop(vr.ATTESTATION_KEYS[0])
        errors, _ = vr.evaluate_roles(self.roles, {"ADJ-1": rec})
        self.assertTrue(errors)

    def test_entropy_custodian_cannot_be_capture_custodian_or_adjudicator(self):
        roles = copy.deepcopy(self.roles)
        r15 = next(r for r in roles["roles"] if r["id"] == "R15")
        r15["status"], r15["registrant"] = "FILLED", {"name": "Ross Buckley"}
        errors, _ = vr.evaluate_roles(roles, {})
        self.assertTrue(any("R15 and R01" in e for e in errors))
        r15["registrant"] = {"name": "A Person"}
        errors, _ = vr.evaluate_roles(roles, {"ADJ-1": adjudicator("ADJ-1", "A Person", "a@example.invalid")})
        self.assertTrue(any("R15 and R06" in e for e in errors))

    def test_replacement_after_packet_opened_invalidates(self):
        a1 = adjudicator("ADJ-1", "A", "a@example.invalid")
        a2 = adjudicator("ADJ-2", "B", "b@example.invalid")
        frozen = {s: vr.sha(json.dumps(r, sort_keys=True).encode()) for s, r in {"ADJ-1": a1, "ADJ-2": a2}.items()}
        a2b = adjudicator("ADJ-2", "C", "c@example.invalid")
        errors, _ = vr.evaluate_roles(self.roles, {"ADJ-1": a1, "ADJ-2": a2b}, packet_opened=True,
                                      frozen_adjudicator_hashes=frozen)
        self.assertTrue(any("after a packet was opened" in e for e in errors))

    def test_adjudicator_slot_cannot_be_filled_in_roles_json(self):
        roles = copy.deepcopy(self.roles)
        r6 = next(r for r in roles["roles"] if r["id"] == "R06")
        r6["status"], r6["registrant"] = "FILLED", {"name": "X"}
        errors, _ = vr.evaluate_roles(roles, {})
        self.assertTrue(errors)

    def test_missing_role_fails_closed(self):
        roles = copy.deepcopy(self.roles)
        roles["roles"] = [r for r in roles["roles"] if r["id"] != "R09"]
        errors, st = vr.evaluate_roles(roles, {})
        self.assertTrue(errors)
        self.assertEqual(st["decision"], "NOT_SEALABLE")


if __name__ == "__main__":
    unittest.main(verbosity=1)
