def slug(s):
    return "".join(ch.lower() if ch.isalnum() else "-" for ch in s).strip("-").replace("--", "-")


def build():
    universe = []
    seen_ids = set()
    for vendor, parent, cat, decision, criterion, unresolved, evidence in C:
        assert cat in CATEGORIES, f"unknown category {cat}"
        assert decision in ("ELIGIBLE", "EXCLUDED")
        if decision == "ELIGIBLE":
            assert criterion is None and not unresolved
        else:
            assert criterion in ("E1", "E2", "E3", "E4", "E5", "E6"), f"bad criterion {criterion}"
        cid = f"{slug(cat)}__{slug(vendor)}"
        assert cid not in seen_ids, f"duplicate id {cid}"
        seen_ids.add(cid)
        rec = {
            "id": cid,
            "vendor": vendor,
            "parent_company": parent,
            "category": cat,
            "provenance": "Source universe U1: curated enumeration of established, publicly-documented "
                          "commercial developer platforms across the seven frozen Programme-A categories, "
                          "compiled from general public knowledge of category-defining API vendors as of "
                          f"{CONSTRUCTION_DATE}. Membership rule is independent of any FIRSTCALL outcome, "
                          "agent difficulty, or documentation quality.",
            "source": "U1",
            "decision": decision,
            "exclusion_criterion": criterion,
            "unresolved_evidence": bool(unresolved),
            "evidence": evidence,
            "evidence_authority": "stable public documentary knowledge (2026-09-18); authoritative "
                                  "per-vendor public-doc capture DEFERRED TO FREEZE 4 per frozen sequence",
        }
        universe.append(rec)
    universe.sort(key=lambda r: r["id"])
    return universe


def validate(universe):
    errs = []
    total = len(universe)
    eligible = [r for r in universe if r["decision"] == "ELIGIBLE"]
    excluded = [r for r in universe if r["decision"] == "EXCLUDED"]
    # conservation
    if total != len(eligible) + len(excluded):
        errs.append("CONSERVATION FAIL: total != eligible + excluded")
    # exactly one terminal decision each
    for r in universe:
        if r["decision"] not in ("ELIGIBLE", "EXCLUDED"):
            errs.append(f"non-terminal decision {r['id']}")
    # no duplicate identity (by id and by (category,vendor))
    ids = [r["id"] for r in universe]
    if len(ids) != len(set(ids)):
        errs.append("duplicate id")
    cv = [(r["category"], r["vendor"]) for r in universe]
    if len(cv) != len(set(cv)):
        errs.append("duplicate (category,vendor)")
    # every eligible has exactly one category in the frozen set
    for r in eligible:
        if r["category"] not in CATEGORIES:
            errs.append(f"eligible {r['id']} bad category")
        if r["exclusion_criterion"] is not None:
            errs.append(f"eligible {r['id']} has exclusion criterion")
    # every exclusion has a frozen criterion
    for r in excluded:
        if r["exclusion_criterion"] not in ("E1", "E2", "E3", "E4", "E5", "E6"):
            errs.append(f"exclusion {r['id']} lacks frozen criterion")
    # prior-FIRSTCALL exclusions follow Freeze 1: every E6 vendor is on the frozen list,
    # and every named frozen E6 real vendor that appears in the universe is excluded E6.
    e6 = [r for r in universe if r["exclusion_criterion"] == "E6"]
    for r in e6:
        if r["vendor"] not in E6_PRIOR_FIRSTCALL:
            errs.append(f"E6 {r['vendor']} not on frozen list")
    present = {r["vendor"] for r in universe}
    for v in E6_PRIOR_FIRSTCALL:
        if v == "CF-001/Acme":
            continue  # internal fixture, not a real external vendor; noted in report, not in U1
        if v in present:
            rr = [r for r in universe if r["vendor"] == v]
            if not all(x["exclusion_criterion"] == "E6" for x in rr):
                errs.append(f"frozen E6 vendor {v} present but not excluded E6")
    # counts
    by_cat = {c: len([r for r in eligible if r["category"] == c]) for c in CATEGORIES}
    by_reason = {}
    for r in excluded:
        by_reason[r["exclusion_criterion"]] = by_reason.get(r["exclusion_criterion"], 0) + 1
    unresolved = [r for r in excluded if r["unresolved_evidence"]]
    return {
        "errors": errs,
        "total": total,
        "eligible": len(eligible),
        "excluded": len(excluded),
        "eligible_by_category": by_cat,
        "excluded_by_reason": by_reason,
        "unresolved_eligibility": len(unresolved),
        "unresolved_ids": sorted(r["id"] for r in unresolved),
        "e6_excluded": sorted(r["vendor"] for r in e6),
    }

    universe = build()
    report = validate(universe)

    eligible = [r for r in universe if r["decision"] == "ELIGIBLE"]
    excluded = [r for r in universe if r["decision"] == "EXCLUDED"]

                           "knowledge as of 2026-09-18; membership independent of FIRSTCALL outcomes, "
                           "agent difficulty, and documentation quality.",
            "reproducibility": "reproducible from the recorded candidate list + frozen criteria + RULE-P; "
                               "no external file was fetched, so no external snapshot hash is claimed",
            "disclosed_bias": "popularity/availability bias inherent to any public developer-platform "
                              "enumeration; disclosed, not corrected; selection within U1 is mechanical",
        },
        "category_system": CATEGORIES,

                "frozen categories) is admitted ONCE in the alphabetically-first frozen category it serves; "
                "every other served category is an explicit E5 exclusion. Alphabetical-first is draw-agnostic.",

                "frozen criterion E2 with unresolved_evidence=true and reported as UNRESOLVED ELIGIBILITY.",

        p = os.path.join(here, name)
        with open(p, "w") as f:
            json.dump(obj, f, indent=2, sort_keys=False)
            f.write("\n")
        return p, hashlib.sha256(open(p, "rb").read()).hexdigest()

    _, uh = write_json("candidate-universe.json", frame_doc)

    # conservation validator artifact (re-reads and re-checks)
    with open(os.path.join(here, "candidate-universe.json")) as f:
        reload = json.load(f)
    re_report = validate(reload["candidate_universe"])
    checks = {
        "conservation_total_eq_eligible_plus_excluded":
            reload["counts"]["total_candidates"] == reload["counts"]["eligible"] + reload["counts"]["excluded"],
        "every_candidate_exactly_one_terminal_decision":

            all(r["exclusion_criterion"] in ("E1", "E2", "E3", "E4", "E5", "E6") for r in reload["exclusions"]),

            all(v in E6_PRIOR_FIRSTCALL for v in re_report["e6_excluded"]),
        "no_selected_vendor_artifact": not any(

            ("selected-vendors.json", "draw.json", "vendor-draw.json", "reserves.json", "seed.txt")),

        "real_vendor_runs_zero": reload["real_vendor_runs"] == 0,
        "validator_no_errors": re_report["errors"] == [],
    }
    validator = {
        "schema": "firstcall.programmeA.frame_validator",
        "frame_version": FRAME_VERSION,
        "candidate_universe_sha256": uh,
        "checks": checks,
        "all_pass": all(checks.values()),
        "counts": reload["counts"],
        "unresolved_ids": re_report["unresolved_ids"],
    }
    write_json("frame-validation.json", validator)
