#!/usr/bin/env python3
"""FIRSTCALL Programme A registered source-discovery capture runner (Freeze 1.2).

Implements the single authoritative Bing capture frozen in Freeze 1.1 §3/§8.2 with the
registration repair of Freeze 1.2. It has three modes:

  --offline-describe   print the frozen request plan; no network at all.
  --preflight-check    verify registration, clock (SNTP) and egress echo; NEVER contacts
                       the discovery source; writes nothing; may be repeated.
  --authoritative      the one registered attempt. Creates the capture directory
                       exclusively (a second invocation is refused and logged), waits
                       without any source contact until the scheduled instant, issues the
                       four queries at the frozen offsets, archives raw bytes before
                       parsing, and writes the session manifest. No retry, no repeat.

The discovery source can only be contacted through LiveEnv.http_get after run_capture
arms the source guard at a scheduled request instant. Offline tests drive the same code
with a fake environment and a socket ban.
"""

import json
import os
import platform

import struct
import subprocess
import sys
import time
import urllib.parse
from datetime import datetime, timezone
from pathlib import Path

RUNNER_ID = "firstcall-programme-a-capture-runner"
RUNNER_VERSION = "1.2.0"
REGISTRATION_TAG = "programme-a-freeze-1.2-registration"
T0_UTC = "2026-09-17T00:00:00Z"
SCHEDULED_START_UTC = "2026-09-19T09:00:00Z"
PREFLIGHT_AT_UTC = "2026-09-19T08:50:00Z"
PREFLIGHT_DEADLINE_UTC = "2026-09-19T08:58:00Z"
QUERIES = (
    "public API directory downloadable dataset",
    "open software API catalog CSV JSON",
    "public developer platform registry bulk export",
    "open API index data repository",
)
OFFSETS_SECONDS = (0, 60, 120, 180)
MAX_START_LATENESS_SECONDS = 5.0
REQUIRED_ORGANIC_POSITIONS = 20
SEARCH_ENDPOINT = "https://www.bing.com/search"
SEARCH_PARAMS = (("setlang", "en"), ("cc", "GB"), ("count", "20"))
USER_AGENT = ("Mozilla/5.0 (compatible; FIRSTCALL-ProgrammeA-CaptureRunner/1.2.0; "
              "one-time four-query research capture)")
REQUEST_HEADERS = (
    ("User-Agent", USER_AGENT),
    ("Accept", "text/html,application/xhtml+xml;q=0.9,*/*;q=0.1"),
    ("Accept-Language", "en-GB,en;q=0.9"),
    ("Accept-Encoding", "identity"),
    ("Connection", "close"),
)
REQUEST_DEADLINE_SECONDS = 30.0
MAX_RESPONSE_BYTES = 5 * 1024 * 1024
MAX_REDIRECTS = 3
REDIRECT_STATUSES = (301, 302, 303, 307, 308)
SOURCE_HOST_SUFFIX = "bing.com"
SNTP_SERVER = "time.apple.com"
SNTP_ATTEMPTS = 5
SNTP_TIMEOUT_SECONDS = 5.0
MAX_CLOCK_OFFSET_SECONDS = 2.0
EGRESS_ECHO_URL = "https://api.ipify.org"
EGRESS_ATTEMPTS = 5
AUX_RETRY_WAIT_SECONDS = 10.0
GIT_ATTEMPTS = 5
CAPTURE_DIR = "experiments/programme-a/capture/authoritative-2026-09-19T0900Z"
ATTEMPT_LOG = "experiments/programme-a/capture/attempts.jsonl"
REGISTRATION_DIR = "experiments/programme-a/registration"
RUNNER_PATH = REGISTRATION_DIR + "/capture_runner.py"
MANIFEST_PATH = REGISTRATION_DIR + "/capture-runner-manifest.json"
CAFFEINATE = "/usr/bin/caffeinate"


class CaptureHalt(Exception):
    def __init__(self, code, detail=""):
        super().__init__(f"{code}: {detail}")
        self.code = code
        self.detail = detail


class SourceGuardError(RuntimeError):
    pass


def canonical(obj):
    return json.dumps(obj, sort_keys=True, separators=(",", ":"),
                      ensure_ascii=False, allow_nan=False).encode("utf-8")


def sha256(data):
    return hashlib.sha256(data).hexdigest()


def parse_utc(text):
    return datetime.strptime(text, "%Y-%m-%dT%H:%M:%SZ").replace(tzinfo=timezone.utc).timestamp()


def iso(epoch):
    return datetime.fromtimestamp(epoch, timezone.utc).strftime("%Y-%m-%dT%H:%M:%S.%fZ")


def query_url(query):
    return SEARCH_ENDPOINT + "?" + urllib.parse.urlencode((("q", query),) + SEARCH_PARAMS)


def is_source_host(host):
    host = (host or "").lower().rstrip(".")
    return host == SOURCE_HOST_SUFFIX or host.endswith("." + SOURCE_HOST_SUFFIX)


def request_plan():
    start = parse_utc(SCHEDULED_START_UTC)
    return [{"index": i + 1, "query": q, "url": query_url(q), "scheduled_utc": iso(start + off),
             "offset_seconds": off} for i, (q, off) in enumerate(zip(QUERIES, OFFSETS_SECONDS))]


# ---------------------------------------------------------------- organic-result parser

class _OrganicParser(html.parser.HTMLParser):
    """Organic position = <li> whose class tokens include b_algo, in document order.
    Its URL = href of the first <a> inside the first <h2> of that item."""

    def __init__(self):
        super().__init__(convert_charrefs=True)
        self.items = []
        self._li_depth = 0
        self._item = None
        self._item_depth = None
        self._in_h2 = False
        self._h2_seen = False

    def handle_starttag(self, tag, attrs):
        attrs = dict(attrs)
        if tag == "li":
            self._li_depth += 1
            classes = (attrs.get("class") or "").split()
            if self._item is None and "b_algo" in classes:
                self._item = {"href": None}
                self._item_depth = self._li_depth
                self._h2_seen = False
        elif self._item is not None and tag == "h2" and not self._h2_seen:
            self._in_h2 = True
        elif self._item is not None and tag == "a" and self._in_h2 and self._item["href"] is None:
            self._item["href"] = attrs.get("href")

    def handle_endtag(self, tag):
        if tag == "h2" and self._in_h2:
            self._in_h2 = False
            self._h2_seen = True
        elif tag == "li":
            if self._item is not None and self._li_depth == self._item_depth:
                self.items.append(self._item)
                self._item = None
                self._item_depth = None
                self._in_h2 = False
            self._li_depth = max(0, self._li_depth - 1)

    def close(self):
        super().close()
        if self._item is not None:
            self.items.append(self._item)
            self._item = None


def decode_result_href(href):
    """Return the destination URL of an organic link, decoding Bing /ck/a wrappers."""
    if not href:
        return None
    parts = urllib.parse.urlsplit(href)
    if parts.scheme not in ("http", "https") or not parts.netloc:
        return None
    if is_source_host(parts.hostname) and parts.path == "/ck/a":
        values = urllib.parse.parse_qs(parts.query, keep_blank_values=True).get("u", [])
        if len(values) != 1 or not values[0].startswith("a1"):
            return None
        payload = values[0][2:]
        try:
            target = base64.urlsafe_b64decode(payload + "=" * (-len(payload) % 4)).decode("utf-8")
        except (ValueError, UnicodeDecodeError):
            return None
        tparts = urllib.parse.urlsplit(target)
        return target if tparts.scheme in ("http", "https") and tparts.netloc else None
    return href


def parse_organic(body):
    parser = _OrganicParser()
    parser.feed(body.decode("utf-8", errors="replace"))
    parser.close()
    out = []
    for position, item in enumerate(parser.items, start=1):
        out.append({"position": position, "href": item["href"],
                    "result_url": decode_result_href(item["href"])})
    return out


def first_positions(results):
    """Positions 1..20 must all exist and be parseable, else the capture stops."""
    head = results[:REQUIRED_ORGANIC_POSITIONS]
    ok = len(head) == REQUIRED_ORGANIC_POSITIONS and all(r["result_url"] for r in head)
    return ok, head


# ---------------------------------------------------------------- environments

class Response:
    def __init__(self, status, headers, body, remote_ip=None, truncated=False):
        self.status = status
        self.headers = headers
        self.body = body
        self.remote_ip = remote_ip
        self.truncated = truncated

    def header(self, name):
        name = name.lower()
        return [v for k, v in self.headers if k.lower() == name]


class _IPv4HTTPSConnection(http.client.HTTPSConnection):
    def connect(self):
        infos = socket.getaddrinfo(self.host, self.port, socket.AF_INET, socket.SOCK_STREAM)
        if not infos:
            raise OSError("no IPv4 address")
        sock = socket.create_connection(infos[0][4], self.timeout)
        self.sock = self._context.wrap_socket(sock, server_hostname=self.host)


class LiveEnv:
    """Real clock/network. Source requests are refused unless armed by run_capture."""

    def __init__(self, root):
        self.root = Path(root)
        self.source_armed_until = None
        self._awake = None

    def now(self):
        return time.time_ns() / 1e9

    def sleep(self, seconds):
        time.sleep(max(0.0, seconds))

    def arm_source(self, until_epoch):
        self.source_armed_until = until_epoch

    def disarm_source(self):
        self.source_armed_until = None

    def http_get(self, url, headers, timeout):
        parts = urllib.parse.urlsplit(url)
        if parts.scheme != "https":
            raise SourceGuardError("https only")
        if is_source_host(parts.hostname):
            if self.source_armed_until is None or self.now() > self.source_armed_until:
                raise SourceGuardError("discovery source contacted outside an armed request")
        conn = _IPv4HTTPSConnection(parts.hostname, parts.port or 443, timeout=timeout,
                                    context=ssl.create_default_context())
        try:
            path = urllib.parse.urlunsplit(("", "", parts.path or "/", parts.query, ""))
            conn.putrequest("GET", path, skip_host=False, skip_accept_encoding=True)
            for k, v in headers:
                conn.putheader(k, v)
            conn.endheaders()
            resp = conn.getresponse()
            remote_ip = conn.sock.getpeername()[0] if conn.sock else None
            body = resp.read(MAX_RESPONSE_BYTES + 1)
            return Response(resp.status, list(resp.getheaders()), body[:MAX_RESPONSE_BYTES],
                            remote_ip, truncated=len(body) > MAX_RESPONSE_BYTES)
        finally:
            conn.close()

    def sntp_offset(self):
        addr = socket.getaddrinfo(SNTP_SERVER, 123, socket.AF_INET, socket.SOCK_DGRAM)[0][4]
        with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as s:
            s.settimeout(SNTP_TIMEOUT_SECONDS)
            t0 = self.now()
            s.sendto(b"\x1b" + 47 * b"\0", addr)
            data, _ = s.recvfrom(512)
            t3 = self.now()
        if len(data) < 48:
            raise OSError("short SNTP reply")
        rs, rf, ts, tf = struct.unpack("!IIII", data[32:48])
        t1 = rs - 2208988800 + rf / 2**32
        t2 = ts - 2208988800 + tf / 2**32
        return {"server": SNTP_SERVER, "server_ip": addr[0], "offset_seconds": ((t1 - t0) + (t2 - t3)) / 2,
                "round_trip_seconds": (t3 - t0) - (t2 - t1), "measured_at_system_utc": iso(t3)}

    def egress_ip(self):
        r = self.http_get(EGRESS_ECHO_URL, (("User-Agent", USER_AGENT), ("Connection", "close")), 15)
        if r.status != 200:
            raise OSError(f"egress echo status {r.status}")
        return str(ipaddress.IPv4Address(r.body.decode("ascii").strip()))

    def git(self, *args):
        return subprocess.run(["git", "-C", str(self.root), *args], check=True,
                              capture_output=True, timeout=60).stdout

    def verify_registration(self):
        own = Path(__file__).resolve()
        own_bytes = own.read_bytes()
        manifest_bytes = (self.root / MANIFEST_PATH).read_bytes()
        manifest = json.loads(manifest_bytes)
        rt = manifest["runtime"]
        exe = os.path.realpath(sys.executable)
        checks = {
            "runner_path_is_registered_path": own == (self.root / RUNNER_PATH).resolve(),
            "runner_sha256_matches_manifest": sha256(own_bytes) == manifest["runner"]["sha256"],
            "interpreter_realpath_matches": exe == rt["interpreter_realpath"],
            "interpreter_sha256_matches": sha256(Path(exe).read_bytes()) == rt["interpreter_sha256"],
            "python_version_matches": sys.version == rt["python_version"],
            "openssl_version_matches": ssl.OPENSSL_VERSION == rt["openssl_version"],
        }
        commit = self.git("rev-parse", REGISTRATION_TAG + "^{commit}").decode().strip()
        remote = None
        for attempt in range(GIT_ATTEMPTS):
            try:
                out = self.git("ls-remote", "origin", f"refs/tags/{REGISTRATION_TAG}^{{}}").decode().split()
                remote = out[0] if out else None
                break
            except (subprocess.SubprocessError, OSError):
                if attempt + 1 < GIT_ATTEMPTS:
                    self.sleep(AUX_RETRY_WAIT_SECONDS)
        checks["tag_remotely_anchored_same_commit"] = remote == commit
        checks["runner_bytes_equal_tagged_bytes"] = self.git("show", f"{commit}:{RUNNER_PATH}") == own_bytes
        checks["manifest_bytes_equal_tagged_bytes"] = self.git("show", f"{commit}:{MANIFEST_PATH}") == manifest_bytes
        checks["registration_dir_clean"] = self.git("status", "--porcelain", "--", REGISTRATION_DIR) == b""
        return {"registration_commit": commit, "remote_peeled_commit": remote, "checks": checks,
                "runner_sha256": sha256(own_bytes), "manifest_sha256": sha256(manifest_bytes),
                "ok": all(checks.values())}

    def keep_awake(self):
        self._awake = subprocess.Popen([CAFFEINATE, "-dimsu", "-w", str(os.getpid())])
        return {"caffeinate": CAFFEINATE, "pid": self._awake.pid}

    def runtime_facts(self):
        exe = os.path.realpath(sys.executable)
        return {"interpreter_realpath": exe, "python_version": sys.version,
                "openssl_version": ssl.OPENSSL_VERSION, "platform": platform.platform(),
                "mac_ver": platform.mac_ver()[0], "machine": platform.machine(),
                "hostname": platform.node()}


# ---------------------------------------------------------------- capture procedure

class Recorder:
    def __init__(self, out_dir, env):
        self.out = Path(out_dir)
        self.env = env
        self.seq = 0

    def write_excl(self, rel, data):
        path = self.out / rel
        path.parent.mkdir(parents=True, exist_ok=True)
        fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o444)
        try:
            os.write(fd, data)
            os.fsync(fd)
        finally:
            os.close(fd)
        return sha256(data)

    def event(self, kind, **fields):
        self.seq += 1
        line = canonical({"seq": self.seq, "event": kind, "system_utc": iso(self.env.now()), **fields}) + b"\n"
        fd = os.open(self.out / "events.jsonl", os.O_WRONLY | os.O_CREAT | os.O_APPEND, 0o644)
        try:
            os.write(fd, line)
            os.fsync(fd)
        finally:
            os.close(fd)


def _retry(env, attempts, fn):
    errors = []
    for i in range(attempts):
        try:
            return fn(), errors
        except Exception as exc:  # recorded, never hidden
            errors.append(f"{type(exc).__name__}: {exc}")
            if i + 1 < attempts:
                env.sleep(AUX_RETRY_WAIT_SECONDS)
    return None, errors


def wait_until(env, target, offset):
    """Sleep in <=1 s chunks on the SNTP-corrected clock; no network activity."""
    while True:
        remaining = target - (env.now() + offset)
        if remaining <= 0:
            return
        env.sleep(min(remaining, 1.0))


def fetch_query(env, rec, index, url, scheduled, offset):
    """One request (plus in-source redirects, max 3). No retry. Returns response dict."""
    corrected_start = env.now() + offset
    lateness = corrected_start - scheduled
    rec.event("request_due", index=index, lateness_seconds=round(lateness, 6))
    if lateness > MAX_START_LATENESS_SECONDS:
        raise CaptureHalt("LATE_START", f"query {index} lateness {lateness:.3f}s")
    system_start = env.now()
    deadline = system_start + REQUEST_DEADLINE_SECONDS
    env.arm_source(deadline)
    hops, cookies, current = [], {}, url
    try:
        while True:
            remaining = deadline - env.now()
            if remaining <= 0:
                raise CaptureHalt("UNAVAILABLE_RESULT_PAGE", f"query {index} exceeded {REQUEST_DEADLINE_SECONDS}s")
            headers = list(REQUEST_HEADERS)
            if cookies:
                headers.append(("Cookie", "; ".join(f"{k}={v}" for k, v in sorted(cookies.items()))))
            hop_start = env.now()
            try:
                resp = env.http_get(current, tuple(headers), remaining)
            except SourceGuardError:
                raise
            except Exception as exc:
                raise CaptureHalt("NETWORK_FAILURE", f"query {index}: {type(exc).__name__}: {exc}")
            hops.append({"url": current, "status": resp.status, "remote_ip": resp.remote_ip,
                         "started_system_utc": iso(hop_start), "ended_system_utc": iso(env.now()),
                         "headers": [[k, v] for k, v in resp.headers],
                         "body_sha256": sha256(resp.body), "body_bytes": len(resp.body),
                         "truncated": resp.truncated})
            for sc in resp.header("Set-Cookie"):
                pair = sc.split(";", 1)[0]
                if "=" in pair:
                    k, v = pair.split("=", 1)
                    cookies[k.strip()] = v.strip()
            if resp.status in REDIRECT_STATUSES:
                loc = resp.header("Location")
                if len(hops) > MAX_REDIRECTS or len(loc) != 1:
                    raise CaptureHalt("UNAVAILABLE_RESULT_PAGE", f"query {index}: redirect limit/ambiguous Location")
                nxt = urllib.parse.urljoin(current, loc[0])
                p = urllib.parse.urlsplit(nxt)
                if p.scheme != "https" or not is_source_host(p.hostname):
                    raise CaptureHalt("UNAVAILABLE_RESULT_PAGE", f"query {index}: redirect leaves source host")
                current = nxt
                continue
            break
    finally:
        env.disarm_source()
    return {"index": index, "request_url": url, "final_url": current, "status": resp.status,
            "system_start_utc": iso(system_start), "system_start_epoch": system_start,
            "corrected_lateness_seconds": lateness, "hops": hops, "body": resp.body,
            "truncated": resp.truncated}


def run_capture(env, out_dir, root_for_log=None):
    """The single authoritative attempt. Returns the session manifest dict."""
    out = Path(out_dir)
    log_path = Path(root_for_log) / ATTEMPT_LOG if root_for_log else out.parent / "attempts.jsonl"
    log_path.parent.mkdir(parents=True, exist_ok=True)
    attempt = {"runner_id": RUNNER_ID, "runner_version": RUNNER_VERSION, "invoked_system_utc": iso(env.now()),
               "pid": os.getpid(), "capture_dir": str(out)}
    try:
        out.mkdir(parents=False, exist_ok=False)
    except FileExistsError:
        with open(log_path, "ab") as fh:
            fh.write(canonical({**attempt, "result": "REFUSED_RERUN_CAPTURE_DIR_EXISTS"}) + b"\n")
        raise CaptureHalt("REFUSED_RERUN", "capture directory already exists; the first attempt is authoritative")
    with open(log_path, "ab") as fh:
        fh.write(canonical({**attempt, "result": "ATTEMPT_STARTED"}) + b"\n")
        fh.flush()
        os.fsync(fh.fileno())
    rec = Recorder(out, env)
    rec.write_excl("attempt.json", canonical(attempt))
    rec.event("attempt_started")
    start = parse_utc(SCHEDULED_START_UTC)
    manifest = {"schema": "firstcall.programmeA.source_capture_session.v1", "runner_id": RUNNER_ID,
                "runner_version": RUNNER_VERSION, "t0_utc": T0_UTC, "scheduled_start_utc": SCHEDULED_START_UTC,
                "queries": list(QUERIES), "offsets_seconds": list(OFFSETS_SECONDS),
                "request_plan": request_plan(), "responses": [], "status": None, "halt": None}
    try:
        if env.now() > parse_utc(PREFLIGHT_DEADLINE_UTC):
            raise CaptureHalt("INVOKED_TOO_LATE", "authoritative invocation after preflight deadline")
        manifest["keep_awake"] = env.keep_awake()
        wait_until(env, parse_utc(PREFLIGHT_AT_UTC), 0.0)
        rec.event("preflight_started")
        try:
            reg = env.verify_registration()
        except Exception as exc:  # e.g. tag absent: recorded, then halts below
            reg = {"ok": False, "error": f"{type(exc).__name__}: {exc}"}
        sntp, sntp_err = _retry(env, SNTP_ATTEMPTS, env.sntp_offset)
        egress, egress_err = _retry(env, EGRESS_ATTEMPTS, env.egress_ip)
        pre = {"registration": reg, "sntp": sntp, "sntp_errors": sntp_err, "egress_ipv4": egress,
               "egress_errors": egress_err, "runtime": env.runtime_facts()}
        pre_hash = rec.write_excl("preflight.json", canonical(pre))
        manifest["preflight_sha256"] = pre_hash
        rec.event("preflight_recorded", sha256=pre_hash)
        if not reg.get("ok"):
            raise CaptureHalt("PREFLIGHT_REGISTRATION_MISMATCH", json.dumps(reg.get("checks") or reg.get("error")))
        if sntp is None or abs(sntp["offset_seconds"]) > MAX_CLOCK_OFFSET_SECONDS:
            raise CaptureHalt("PREFLIGHT_CLOCK_UNVERIFIED", json.dumps(sntp))
        if egress is None:
            raise CaptureHalt("PREFLIGHT_EGRESS_UNVERIFIED", "; ".join(egress_err))
        if env.now() > parse_utc(PREFLIGHT_DEADLINE_UTC):
            raise CaptureHalt("PREFLIGHT_OVERRAN_DEADLINE", "")
        offset = sntp["offset_seconds"]
        for i, (query, off) in enumerate(zip(QUERIES, OFFSETS_SECONDS), start=1):
            scheduled = start + off
            wait_until(env, scheduled, offset)
            resp = fetch_query(env, rec, i, query_url(query), scheduled, offset)
            body = resp.pop("body")
            raw_hash = rec.write_excl(f"raw/q{i}.html", body)
            meta = {k: v for k, v in resp.items()}
            meta["raw_sha256"] = raw_hash
            meta["raw_bytes"] = len(body)
            rec.write_excl(f"raw/q{i}.meta.json", canonical(meta))
            rec.event("response_archived", index=i, raw_sha256=raw_hash, status=resp["status"])
            manifest["responses"].append(meta)
            if resp["status"] != 200 or resp["truncated"]:
                raise CaptureHalt("UNAVAILABLE_RESULT_PAGE", f"query {i} status {resp['status']} truncated {resp['truncated']}")
            ok, head = first_positions(parse_organic(body))
            meta["organic_positions"] = head
            rec.event("response_parsed", index=i, parseable_positions=sum(1 for r in head if r["result_url"]))
            if not ok:
                raise CaptureHalt("INSUFFICIENT_ORGANIC_POSITIONS", f"query {i}")
        post_sntp, post_sntp_err = _retry(env, SNTP_ATTEMPTS, env.sntp_offset)
        post_egress, post_egress_err = _retry(env, EGRESS_ATTEMPTS, env.egress_ip)
        manifest["postflight"] = {"sntp": post_sntp, "sntp_errors": post_sntp_err,
                                  "egress_ipv4": post_egress, "egress_errors": post_egress_err}
        if post_sntp is not None:
            for meta in manifest["responses"]:
                scheduled = start + OFFSETS_SECONDS[meta["index"] - 1]
                late_post = meta["system_start_epoch"] + post_sntp["offset_seconds"] - scheduled
                meta["post_offset_lateness_seconds"] = late_post
                if late_post > MAX_START_LATENESS_SECONDS:
                    raise CaptureHalt("LATE_START_POST_CHECK", f"query {meta['index']} {late_post:.3f}s")
        if post_egress is not None and post_egress != egress:
            raise CaptureHalt("EGRESS_CHANGED", f"{egress} -> {post_egress}")
        manifest["egress_post_verification"] = "MATCH" if post_egress is not None else "UNVERIFIED_POSTFLIGHT"
        manifest["status"] = "CAPTURED_COMPLETE"
    except CaptureHalt as halt:
        manifest["status"] = "HALTED"
        manifest["halt"] = {"code": halt.code, "detail": halt.detail}
        rec.event("halt", code=halt.code, detail=halt.detail)
    except Exception as exc:  # any unexpected fault is a recorded, final halt
        manifest["status"] = "HALTED"
        manifest["halt"] = {"code": "RUNNER_EXCEPTION", "detail": f"{type(exc).__name__}: {exc}"}
        rec.event("halt", code="RUNNER_EXCEPTION", detail=manifest["halt"]["detail"])
    finally:
        env.disarm_source()
    manifest["ordered_raw_sha256"] = [m["raw_sha256"] for m in manifest["responses"]]
    data = canonical(manifest)
    manifest_hash = rec.write_excl("session-manifest.json", data)
    rec.event("session_manifest_written", sha256=manifest_hash, status=manifest["status"])
    with open(log_path, "ab") as fh:
        fh.write(canonical({**attempt, "result": manifest["status"], "session_manifest_sha256": manifest_hash}) + b"\n")
    manifest["_session_manifest_sha256"] = manifest_hash
    return manifest


def preflight_check(env):
    """Repeatable readiness check. Never contacts the discovery source; writes nothing."""
    try:
        reg = env.verify_registration()
    except Exception as exc:
        reg = {"ok": False, "error": f"{type(exc).__name__}: {exc}"}
    sntp, sntp_err = _retry(env, 1, env.sntp_offset)
    egress, egress_err = _retry(env, 1, env.egress_ip)
    return {"mode": "preflight-check", "source_contacted": False, "registration": reg, "sntp": sntp,
            "sntp_errors": sntp_err, "egress_ipv4": egress, "egress_errors": egress_err,
            "runtime": env.runtime_facts()}


def main(argv=None):
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    mode = ap.add_mutually_exclusive_group(required=True)
    mode.add_argument("--offline-describe", action="store_true")
    mode.add_argument("--preflight-check", action="store_true")
    mode.add_argument("--authoritative", action="store_true")
    args = ap.parse_args(argv)
    root = Path(__file__).resolve().parents[3]
    if args.offline_describe:
        print(json.dumps({"runner_id": RUNNER_ID, "runner_version": RUNNER_VERSION, "t0_utc": T0_UTC,
                          "preflight_at_utc": PREFLIGHT_AT_UTC, "preflight_deadline_utc": PREFLIGHT_DEADLINE_UTC,
                          "request_plan": request_plan(), "headers": REQUEST_HEADERS,
                          "capture_dir": CAPTURE_DIR}, indent=2))
        return 0
    env = LiveEnv(root)
    if args.preflight_check:
        print(json.dumps(preflight_check(env), indent=2, default=str))
        return 0
    try:
        manifest = run_capture(env, root / CAPTURE_DIR, root_for_log=root)
    except CaptureHalt as halt:
        print(json.dumps({"status": "HALTED", "code": halt.code, "detail": halt.detail}))
        return 3
    print(json.dumps({"status": manifest["status"], "halt": manifest["halt"],
                      "session_manifest_sha256": manifest["_session_manifest_sha256"],
                      "ordered_raw_sha256": manifest["ordered_raw_sha256"]}, indent=2))
    return 0 if manifest["status"] == "CAPTURED_COMPLETE" else 2


if __name__ == "__main__":
    sys.exit(main())
